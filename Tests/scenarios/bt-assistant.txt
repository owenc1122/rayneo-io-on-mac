# Virtual phone connected; double-click the crown to open the AI assistant (WAKEUP on 0x0d); the phone
# drives the session and answers. Duration arg covers boot+connect. A retry double-click after the first
# capture keeps it reliable when the host is CPU-starved and the first gesture is missed.
emulation RunFor "30"
gpio0 Pulse 1 false 120
emulation RunFor "3"
gpio0 Pulse 1 false 120
emulation RunFor "0.2"
gpio0 Pulse 1 false 120
emulation RunFor "5"
lcdc SaveFrame @OUT/shot-ai-1.pgm
gpio0 Pulse 1 false 120
emulation RunFor "0.2"
gpio0 Pulse 1 false 120
emulation RunFor "5"
lcdc SaveFrame @OUT/shot-ai-2.pgm
emulation RunFor "4"
lcdc SaveFrame @OUT/shot-ai-3.pgm
